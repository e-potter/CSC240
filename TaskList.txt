public class TaskList {

	public static void main(String[] args) 
	{

		Task t1, t2, t3, t4, t5;
		
		t1 = new Task("Attend Class");
		t1.setPriority(1);
		
		t2 = new Task("Homework");
		t2.setPriority(2);
		
		t3 = new Task("Exercise");
		t3.setPriority(3);
		
		t4 = new Task("Eat Breakfast");
		t4.setPriority(4);
		
		t5 = new Task("Eat Lunch");
		t5.setPriority(5);
		
		System.out.println("Priority Task List");
		System.out.println("------------------");
		System.out.println(t1.getTask() + "\t prioirty: " + t1.getPriority());
		System.out.println(t2.getTask() + "\t prioirty: " +t2.getPriority());
		System.out.println(t3.getTask() + "\t prioirty: " +t3.getPriority());
		System.out.println(t4.getTask() + "\t prioirty: " +t4.getPriority());
		System.out.println(t5.getTask() + "\t prioirty: " +t5.getPriority());
		
		
	}

}
