//	Manages students and scores.



public class Gradebook 
{
	// variables
		Student[] student = new Student[5];
		
		//constructor
		public Gradebook()
		{
		}
		
		public Gradebook(Student[] input)
		{
			student = input;
		}
		
		
		//method
		public void addStudent(Student a, int index)
		{
			student[index] = a;
		}
		
		public void sortStudents()
		{
			Sorting.selectionSort(student);
		}
		
		public String toString()
		{
			String result = "";
			
			int index = 0;
			while (index < 5)
				{
				result += student[index];
				index++;
				}
			
			return result;
		}	
	
}
