//	A rational number that uses Euclid's algorithm to find a GCD.


public class RationalNumber {
	
	private int numerator, denominator;
	
	public RationalNumber ()
	{
		numerator = 1;
		denominator = 1;
	}
	
	RationalNumber (int a, int b)
	{
		numerator = a;
		denominator = b;
	}
	
	public int getNumerator()
	{
		return numerator;
	}
	
	public int getDenominator()
	{
		return denominator;
	}
	
	public void setNumerator(int n)
	{
		numerator = n;
	}
	
	public void setDenominator(int d)
	{
		denominator = d;
	}
	
	public int gcd(int num1, int num2)
	{
		while (num1 != num2)
		{
			if (num1 > num2)
				num1 = num1 - num2;
			else
				num2 = num2 - num1;
		}
		
		return num1;
	}
	
	public String toString()
	{
		String result;
		int gcd;
		
		gcd = gcd(numerator, denominator);
		
		result = "\nGreatest common divsor of " + numerator + " and " + denominator + " is " + gcd;
		
		return result;
	}
	
}
