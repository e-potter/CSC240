//	Class of a single food item.

public class Food 
{
	private String name, type;
	private int calories;
	private double percent;
	
	public Food()
	{
		name = "";
		type = "";
		calories = 0;
		percent = 0;
	}
	
	public Food(String n, String t, int c, double p)
	{
		name = n;
		type = t;
		calories = c;
		percent = p;
	}
	
	public void setName(String n)
	{
		name = n;
	}
	public void setType(String t)
	{
		type = t;
	}
	public void setCalories(int c)
	{
		calories = c;
	}
	
	public void setPercent(int p)
	{
		percent = p;
	}
	
	public String getName()
	{
		return name;
	}
	public String getType()
	{
		return type;
	}
	public int getCalories()
	{
		return calories;
	}
	public double getPercent()
	{
		return percent;
	}
	
	public String toString()
	{
		String FOOD = name + "\t\t\t" + type + "\t\t\t" + calories + "\t\t\t" + percent;
		return FOOD;
	}
	
}
